可重建 harness 的步骤如下（在本线工作树根运行）：

```python
from pathlib import Path
root = Path('apps/ios/.build/outcome-check')
root.mkdir(parents=True, exist_ok=True)
(root / 'Package.swift').write_text('''// swift-tools-version: 6.0
import PackageDescription
let package = Package(name: "OutcomeMobileChecks", platforms: [.macOS(.v14)], targets: [
 .target(name: "OutcomeGoCore", path: "Sources", resources: [.process("Resources")]),
 .testTarget(name: "OutcomeGoCoreTests", dependencies: ["OutcomeGoCore"], path: "Tests", resources: [.copy("Fixtures")])
])
''')
if not (root / 'Sources').exists():
    (root / 'Sources').symlink_to(Path('apps/ios/Sources/Core').resolve(), target_is_directory=True)
(root / 'Tests').mkdir(exist_ok=True)
for source in Path('apps/ios/Tests/CoreTests').glob('*.swift'):
    if 'OpenTeamCore' in source.read_text():
        continue
    target = root / 'Tests' / source.name
    if not target.exists():
        target.symlink_to(source.resolve())
for target, source in [
    (root / 'Tests/Fixtures', Path('apps/ios/Tests/CoreTests/Fixtures')),
    (Path('apps/ios/.build/Resources'), Path('apps/ios/Resources')),
]:
    if not target.exists():
        target.symlink_to(source.resolve(), target_is_directory=True)
```
